useful commands ::
	docker ps -q -a :: all current containers
		docker stop $(docker ps -q -a) :: stop running containers
		docker rm $(docker ps -a -q) :: delete all current containers
	docker images -a :: all current images
		docker image rm [image name:tag] :: delete image:tag
	docker volume ls :: all current volumes
		docker volume rm [volume name] :: delete volume

Dockerfile nginx ::
	apt update -y && apt install nginx -y && apt install openssl
	COPY my.conf in /etc/nginx/conf.d/ ::
		server {
			listen 80;
			server_name inception;
			return 301 https://$host$request_uri;
		}
		server {
			listen 443 ssl;
			listen [::]:443 ssl;
			
			server_name ;
			ssl_certificate ;
			ssl_certificate_key ;
			ssl_protocols TLSv1.2 TLSv1.3;
			ssl_ciphers HIGH:!aNULL:!MD5;
			
			root /var/www/html;
			index index.php index.html index.htm;
			
			location / {
				try_files $uri $uri/ =404;
		    	}

			location ~ \.php$ {
				fastcgi_pass wordpress:9000;

		}
